package com.jamal2367.styx.database

import android.os.Parcelable
import kotlinx.parcelize.Parcelize

/**
 * A sealed class representing a bookmark item. It is either a bookmark [Entry]
 * pointing to a URL, or a [Folder] grouping entries together.
 */
sealed class Bookmark {

    /**
     * A concrete bookmark with a [url], [title], [position] in its folder,
     * and the [folder] it belongs to.
     */
    @Parcelize
    data class Entry(
        val url: String,
        val title: String,
        val position: Int,
        val folder: Folder
    ) : Bookmark(), Parcelable

    /**
     * A bookmark folder identified by its [title]. An empty title represents
     * the root (default) folder.
     */
    @Parcelize
    data class Folder(
        val title: String
    ) : Bookmark(), Parcelable

}

/**
 * Converts a nullable [String] to a [Bookmark.Folder].
 * A null or blank value maps to the root folder (empty title).
 */
fun String?.asFolder(): Bookmark.Folder = Bookmark.Folder(this ?: "")
